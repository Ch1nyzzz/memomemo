\documentclass[border=2pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc, automata}

\definecolor{statefill}{HTML}{EEEDFE}
\definecolor{stateborder}{HTML}{534AB7}
\definecolor{statetext}{HTML}{3C3489}
\definecolor{escalate}{HTML}{5F5E5A}
\definecolor{resetgreen}{HTML}{265B19}
\definecolor{arrowgray}{HTML}{737268}

\begin{document}
\begin{tikzpicture}[
    >=Stealth,
    node distance=3.2cm,
    every node/.style={font=\small},
    state/.style={draw=stateborder, fill=statefill, rounded corners=6pt,
        minimum height=1.8cm, minimum width=2.8cm, align=center},
    arw/.style={->, thick, escalate},
    resetarw/.style={->, thick, resetgreen, dashed},
]

% States
\node[state] (low) {
    \textbf{\color{statetext}low}\\[3pt]
    {\scriptsize\color{stateborder} trace: last1}\\
    {\scriptsize\color{stateborder} refs: best1 + worst}
};
\node[state, right=of low] (medium) {
    \textbf{\color{statetext}medium}\\[3pt]
    {\scriptsize\color{stateborder} trace: last3}\\
    {\scriptsize\color{stateborder} refs: best3 + worst}
};
\node[state, right=of medium] (high) {
    \textbf{\color{statetext}high}\\[3pt]
    {\scriptsize\color{stateborder} trace: all}\\
    {\scriptsize\color{stateborder} refs: full history}
};

% Entry arrow
\draw[->, thick, arrowgray] ($(low.north) + (0,0.8)$) -- (low.north)
    node[pos=0, above, font=\scriptsize\color{arrowgray}] {iter $\leq$ 5: forced};

% Escalation arrows (stagnation)
\draw[arw] (low.east) -- (medium.west)
    node[midway, above, font=\scriptsize] {stagnation};
\draw[arw] (medium.east) -- (high.west)
    node[midway, above, font=\scriptsize] {stagnation};

% Self-loop on high
\draw[arw] (high.north east) .. controls ++(0.8,0.8) and ++(0.8,-0.8) .. (high.south east)
    node[pos=0.5, right, font=\scriptsize, align=left] {stagnation\\(stay)};

% Reset arrows (improvement)
\draw[resetarw] (medium.south) .. controls ++(0,-0.8) and ++(0,-0.8) .. (low.south)
    node[pos=0.5, below, font=\scriptsize\color{resetgreen}] {improved};
\draw[resetarw] (high.south) .. controls ++(0,-1.4) and ++(0,-1.4) .. (low.south east)
    node[pos=0.5, below, font=\scriptsize\color{resetgreen}] {improved};

% Legend
\node[below=2.0cm of medium, font=\scriptsize, align=left] (legend) {
    \tikz\draw[->, thick, escalate] (0,0) -- (0.6,0); stagnation $\to$ escalate budget \qquad
    \tikz\draw[->, thick, resetgreen, dashed] (0,0) -- (0.6,0); improved $\to$ reset to \texttt{low}
};

\node[below=0.15cm of legend, font=\scriptsize\color{arrowgray}] {
    ``Improved'' = new entry on Pareto frontier
};

\end{tikzpicture}
\end{document}
